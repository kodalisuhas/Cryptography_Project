## Key Interfaces I Will Implement

### Padding (src/utils/padding.py)
class Padding:
    @staticmethod
    def pad(data: bytes, block_size: int = 16) -> bytes
    @staticmethod
    def unpad(data: bytes) -> bytes
    @staticmethod
    def is_padded(data: bytes) -> bool

### AESCipher (src/core/aes_cipher.py)
class AESCipher:
    VALID_KEY_SIZES = [128, 192, 256]
    VALID_MODES = ['ECB', 'CBC']
    BLOCK_SIZE = 16
    
    def __init__(self, key_size: int, mode: str)
    def generate_key(self) -> bytes
    def generate_iv(self) -> bytes
    def encrypt(self, plaintext: bytes, key: bytes, iv: bytes = None) -> bytes
    def decrypt(self, ciphertext: bytes, key: bytes, iv: bytes = None) -> bytes
